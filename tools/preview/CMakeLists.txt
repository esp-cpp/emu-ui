# Render emu-ui's screens on a desktop, without a board:
#   cmake -B build -DLVGL_DIR=<path to lvgl 9.6> && cmake --build build && ./build/preview out/
# (LVGL from an esp-box-emu / tab5-emu build: managed_components/lvgl__lvgl)
cmake_minimum_required(VERSION 3.20)
project(emu_ui_preview C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_C_STANDARD 11)
if(NOT LVGL_DIR)
  message(FATAL_ERROR "set LVGL_DIR")
endif()
file(GLOB_RECURSE LVGL_SRCS ${LVGL_DIR}/src/*.c)
add_library(lvgl STATIC ${LVGL_SRCS})
target_include_directories(lvgl PUBLIC ${LVGL_DIR} ${LVGL_DIR}/include ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE)
target_compile_options(lvgl PRIVATE -w)
file(GLOB EMU_UI_SRCS ${CMAKE_CURRENT_SOURCE_DIR}/../../src/*.cpp)
add_executable(preview preview.cpp ${EMU_UI_SRCS})
target_include_directories(preview PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../include)
target_link_libraries(preview lvgl)
